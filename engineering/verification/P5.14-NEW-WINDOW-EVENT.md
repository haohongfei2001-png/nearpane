## 复验

在同意短时显示两个自有后台窗的图形会话中运行以下命令。现有辅助功能权限不足、应用不在活动会话或焦点异常时，测试会拒绝。验收结果检查须包含 JSON 内的字段。

```bash
set -eu
NEARPANE_FOUNDATION_ONLY=1 NEARPANE_OFFLINE=1 scripts/test.sh --filter OrdinaryWindowManagerTests
d=$(mktemp -d /private/tmp/nearpane-owned-creation.XXXXXXXX)
app="$d/Witness.app"
mkdir -p "$app/Contents/MacOS"
cat > "$app/Contents/Info.plist" <<'PLIST'
<?xml version="1.0" encoding="UTF-8"?>
<plist version="1.0"><dict>
<key>CFBundleIdentifier</key><string>org.nearpane.acceptance.NewWindowEvent</string>
<key>CFBundleExecutable</key><string>Witness</string><key>CFBundlePackageType</key><string>APPL</string>
<key>CFBundleVersion</key><string>1</string><key>CFBundleName</key><string>Nearpane synthetic window event</string>
<key>LSUIElement</key><true/><key>NSPrincipalClass</key><string>NSApplication</string>
</dict></plist>
PLIST
dbg=.build/native-foundation/arm64-apple-macosx/debug
xcrun swiftc -module-cache-path "$d/modules" -parse-as-library -I "$dbg/Modules" \
  -Xcc "-fmodule-map-file=$PWD/$dbg/CReadOnlySpaces.build/module.modulemap" \
  scripts/acceptance/ordinary-windows/CreatedEvent.swift \
  Sources/NearpaneOrdinaryWindowController/{CreatedWindowWatch,SystemEvidence,OwnerIdentity}.swift \
  "$dbg"/{NearpaneCore,WindowGeometryCore}.build/*.swift.o "$dbg/CReadOnlySpaces.build/ReadOnlySpaces.c.o" \
  -o "$app/Contents/MacOS/Witness"
codesign --force --sign - "$app"
codesign --verify --strict "$app"
"$app/Contents/MacOS/Witness" > "$d/result.json"
python3 - "$d/result.json" <<'PY'
import json, sys
r = json.load(open(sys.argv[1]))
assert r["productionEventExact"] and r["focusReturned"] and r["externalChildExited"]
assert r["ownerResidual"] == r["AXWrites"] == 0 and r["old"] != r["new"]
e = r["external"]
assert e["oldAbsentBeforeCreation"] and e["existingAX"] == [r["old"]]
assert e["productionWatchTakenIDs"] == e["rawAXWindowCreatedIDs"] == [r["new"]]
assert set(e["finalAX"]) == set(e["independentCGVisible"]) == {r["old"], r["new"]}
PY
```
